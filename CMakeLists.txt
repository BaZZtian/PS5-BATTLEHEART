cmake_minimum_required(VERSION 3.25)
project(BattleHeart VERSION 0.0.1 LANGUAGES C)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/projects.json" PORT_CONFIGURATION)
string(JSON PORT_ID GET "${PORT_CONFIGURATION}" projects 0 id)
string(JSON TEST_DISC_ID GET "${PORT_CONFIGURATION}" projects 0 disc_ids NTSC-U)

add_library(gc_disc STATIC runtime/gc_disc.c)
target_include_directories(gc_disc PUBLIC runtime)
target_compile_features(gc_disc PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(gc_disc PRIVATE -Wall -Wextra -Werror -Wconversion)
endif()

if(PORT_ID STREQUAL "sonic-adventure-2-battle")
  add_library(sonic_gc_media STATIC sonic/sonic_gc_media.c)
  target_link_libraries(sonic_gc_media PUBLIC gc_disc)
  target_include_directories(sonic_gc_media PUBLIC sonic)
  target_compile_features(sonic_gc_media PUBLIC c_std_11)
endif()

if(NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  add_executable(gc_disc_tests tests/gc_disc_tests.c)
  target_link_libraries(gc_disc_tests PRIVATE gc_disc)
  target_compile_definitions(gc_disc_tests PRIVATE PS5_PORTS_TEST_DISC_ID="${TEST_DISC_ID}")
  if(PORT_ID STREQUAL "sonic-adventure-2-battle")
    target_link_libraries(gc_disc_tests PRIVATE sonic_gc_media)
    target_compile_definitions(gc_disc_tests PRIVATE PS5_PORTS_WITH_SONIC=1)
  endif()
  add_test(NAME gc_disc_bounds_and_regions COMMAND gc_disc_tests)
  add_executable(gc_inspect tools/gc_inspect.c)
  target_link_libraries(gc_inspect PRIVATE gc_disc)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME regional_release_gate COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_release_gate.py)
  add_test(NAME regional_preparation COMMAND ${CMAKE_COMMAND} -E env
    GC_INSPECT=$<TARGET_FILE:gc_inspect>
    ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_prepare_region.py)
endif()
